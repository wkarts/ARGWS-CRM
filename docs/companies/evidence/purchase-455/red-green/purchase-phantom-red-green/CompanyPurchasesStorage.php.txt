<?php

declare(strict_types=1);

/** Shared transaction and readback rules for the finite purchasing service only. */
trait CompanyPurchasesStorage
{
    private array $frames = [];
    private array $validators = [];
    private ?array $api = null;
    private ?array $portal = null;

    private static function now(): string { return gmdate('Y-m-d H:i:s'); }
    private function rows(string $sql, array $args = []): array { return IssuerStorage::query($this->db, $sql, $args)->result_array(); }

    public function tables(): array
    {
        $extra = $this->api === null ? [] : ['company_api_tokens','company_api_token_requests'];
        return array_values(array_unique(array_merge(array_keys(CompanyPurchasesSchema::definitions($this->p)), (new CompanyExpenses($this->db, $this->p))->tables(), $extra)));
    }

    private function authorize(int $issuer, int $actor, string $cap, string $operation): void
    {
        if (!$this->acl->can($issuer, $actor, 'purchases', 'view') || !$this->acl->can($issuer, $actor, 'purchases', $this->portal === null ? $cap : 'portal')) { throw new DomainException('company_purchase_denied'); }
        if ($this->api !== null) {
            if ($this->api['issuer_id'] !== $issuer || $this->api['actor_id'] !== $actor) { throw new DomainException('company_api_scope_denied'); }
            $scope=match($operation) {
                'read','list','configuration','files','download','order_pdf'=>'purchases:read',
                'supplier_saved','request_created','quote_created','file_uploaded','file_hidden'=>'purchases:write',
                'request_approved','request_cancelled','order_created','order_cancelled'=>'purchases:approve',
                'received'=>'purchases:receive','document_posted'=>'purchases:post',
                'settlement_declared','settlement_reversed'=>'purchases:pay',
                'invite_created','invite_revoked','invites'=>'purchases:portal',
                'report','export'=>'purchases:report','configured'=>'purchases:settings',
                default=>throw new DomainException('company_api_scope_denied')
            };
            if ($scope!==$this->api['scope']) { throw new DomainException('company_api_scope_denied'); }
            (new CompanyApiTokens($this->db, $this->p))->guard($this->api['secret'], $this->api['scope'], $issuer, $actor);
        }
        if ($this->portal !== null) { $this->guardPortal($issuer, $actor, $operation); }
    }

    private function transaction(int $issuer, int $actor, string $cap, string $operation, ?string $key, callable $work)
    {
        CompanyPurchasesSchema::assertValid($this->db, $this->p);
        $this->frames = []; $this->validators = [];
        try {
            return IssuerTransaction::run($this->db, $this->p, $this->tables(), function () use ($issuer, $actor, $cap, $operation, $key, $work) {
                $this->acl->lockFinancialIdentity($issuer, $actor);
                $this->authorize($issuer, $actor, $cap, $operation);
                $result = $work();
                $this->authorize($issuer, $actor, $cap, $operation);
                if ($this->api !== null) { (new CompanyApiTokens($this->db, $this->p))->usedInOwnedTransaction($this->api['secret'], $this->api['scope'], $issuer, $actor, 'purchases_'.$operation, $key); }
                foreach ($this->validators as $check) { $check(); }
                $this->authorize($issuer, $actor, $cap, $operation);
                foreach ($this->frames as $frame) { $this->confirmFrame($frame); }
                return $result;
            });
        } finally { $this->frames = []; $this->validators = []; }
    }

    private function confirm(string $table, array $where, array $want): void
    {
        $parts = []; foreach ($where as $field => $v) { $parts[] = '`'.$field.'`=?'; }
        $fields = '`'.implode('`,`', array_keys($want)).'`';
        $rows = $this->rows('SELECT '.$fields.' FROM `'.$this->p.$table.'` WHERE '.implode(' AND ', $parts), array_values($where));
        if (count($rows) !== 1) { throw new RuntimeException('issuer_purchase_write_unconfirmed'); }
        foreach ($want as $f => $v) {
            if (!array_key_exists($f, $rows[0]) || ($v === null ? $rows[0][$f] !== null : $rows[0][$f] === null || (string)$rows[0][$f] !== (string)$v)) { throw new RuntimeException('issuer_purchase_write_unconfirmed'); }
        }
    }

    private function remember(string $table, array $row, string $pk = 'id', bool $replace = false): void
    {
        $key = $table.':'.$row[$pk]; $where = [$pk=>$row[$pk]];
        if (isset($this->frames[$key]) && !$replace) { $this->confirmFrame($this->frames[$key]); }
        $this->confirm($table, $where, $row);
        $this->frames[$key] = [$table,$where,array_keys($row),self::frameHash($row)];
    }

    private static function frameHash(array $row): string
    {
        ksort($row); $hash=hash_init('sha256');
        foreach ($row as $field=>$value) { hash_update($hash,strlen($field).':'.$field.($value===null?'N':('S'.strlen((string)$value).':'.(string)$value))); }
        return hash_final($hash);
    }

    /** Keep digests instead of copies of private BLOBs/large snapshots within the 128 MiB runtime. */
    private function confirmFrame(array $frame): void
    {
        [$table,$where,$fields,$hash]=$frame; $parts=[]; foreach ($where as $field=>$v) { $parts[]='`'.$field.'`=?'; }
        $rows=$this->rows('SELECT `'.implode('`,`',$fields).'` FROM `'.$this->p.$table.'` WHERE '.implode(' AND ',$parts),array_values($where));
        if (count($rows)!==1||!hash_equals($hash,self::frameHash($rows[0]))) { throw new RuntimeException('issuer_purchase_write_unconfirmed'); }
    }

    private function insert(string $table, array $data, string $pk = 'id')
    {
        IssuerStorage::query($this->db, 'INSERT INTO `'.$this->p.$table.'` (`'.implode('`,`', array_keys($data)).'`) VALUES ('.implode(',', array_fill(0, count($data), '?')).')', array_values($data));
        $id = $data[$pk] ?? (int)$this->db->insert_id();
        if ($id === 0 || $id === '') { throw new RuntimeException('issuer_purchase_write_unconfirmed'); }
        $this->remember($table, [$pk=>$id] + $data, $pk);
        return $id;
    }

    private function update(string $table, array $old, array $changes, string $pk = 'id'): array
    {
        $parts = []; foreach ($changes as $f => $v) { $parts[] = '`'.$f.'`=?'; }
        IssuerStorage::query($this->db, 'UPDATE `'.$this->p.$table.'` SET '.implode(',', $parts).' WHERE `'.$pk.'`=? AND issuer_id=?', array_merge(array_values($changes), [$old[$pk],$old['issuer_id']]));
        $row = array_replace($old, $changes); $this->remember($table, $row, $pk, true); return $row;
    }

    private function object(int $issuer, string $kind, int $id): array
    {
        $table = self::table($kind);
        $row = $this->rows('SELECT * FROM `'.$this->p.$table.'` WHERE id=? AND issuer_id=? FOR UPDATE', [$id,$issuer])[0] ?? null;
        if (!$row) { throw new DomainException('company_purchase_denied'); }
        if (isset($row['record_hash']) && !hash_equals($row['record_hash'], CompanyPurchasesValues::hash($row))) { throw new RuntimeException('issuer_purchase_integrity_invalid'); }
        if ($kind === 'order' && !hash_equals($row['snapshot_hash'], hash('sha256', $row['snapshot_json']))) { throw new RuntimeException('issuer_purchase_integrity_invalid'); }
        $this->remember($table, $row); return $row;
    }

    private static function table(string $kind): string
    {
        return 'company_purchase_'.match ($kind) { 'supplier'=>'suppliers','request'=>'requests','quote'=>'quotes','order'=>'orders','receipt'=>'receipts','document'=>'documents','payable'=>'payables','event'=>'events','file'=>'files','invite'=>'invites',default=>throw new DomainException('company_purchase_invalid') };
    }

    private function revision(array $row, $revision): void
    {
        if ((int)$row['revision'] !== CompanyPurchasesValues::integer($revision, 1) || (int)$row['revision'] >= 2147483647) { throw new DomainException('company_purchase_stale'); }
    }

    private function command($issuer, $actor, string $cap, string $op, string $key, array $input, callable $work)
    {
        $issuer = IssuerPolicy::id($issuer); $actor = IssuerPolicy::id($actor);
        if (!preg_match('/^[a-f0-9]{64}$/D', $key)) { throw new DomainException('company_purchase_invalid'); }
        $origin = $this->api === null ? null : [$this->api['token_id'],$this->api['scope']];
        $portal = $this->portal === null ? null : [$this->portal['id'],$this->portal['revision']];
        $hash = hash('sha256', CompanyPurchasesValues::json([$op,$input,$origin,$portal]));
        return $this->transaction($issuer, $actor, $cap, $op, $key, function () use ($issuer, $actor, $op, $key, $hash, $work) {
            $old = $this->rows('SELECT * FROM '.$this->p.'company_purchase_commands WHERE request_key=? FOR UPDATE', [$key])[0] ?? null;
            if ($old) {
                if ((int)$old['issuer_id'] !== $issuer || (int)$old['actor_id'] !== $actor || $old['operation'] !== $op || !hash_equals($old['input_hash'], $hash)) { throw new DomainException('company_purchase_request_conflict'); }
                $expectedKind=match($op) { 'configured'=>'settings','supplier_saved'=>'supplier','request_created','request_approved','request_cancelled'=>'request','quote_created'=>'quote','order_created','order_cancelled'=>'order','received'=>'receipt','document_posted'=>'payable','settlement_declared','settlement_reversed'=>'event','file_uploaded','file_hidden'=>'file','invite_created','invite_revoked'=>'invite',default=>throw new RuntimeException('issuer_purchase_receipt_invalid') };
                $result=json_decode($old['result_json'],true,32,JSON_THROW_ON_ERROR);
                $resultHash=hash('sha256',CompanyPurchasesValues::json([$issuer,$actor,$op,$hash,$old['object_kind'],(int)$old['object_id'],$old['result_json']]));
                if ($old['object_kind']!==$expectedKind||!is_int($result)||$result<1||$result!==(int)$old['object_id']||!hash_equals($old['result_hash'],$resultHash)) { throw new RuntimeException('issuer_purchase_receipt_invalid'); }
                if ($expectedKind === 'settings') { if ($result!==$issuer) { throw new RuntimeException('issuer_purchase_receipt_invalid'); } $this->settingsRow($issuer); }
                else {
                    $r = $this->object($issuer, $expectedKind, $result);
                    if ($expectedKind === 'payable') { $this->payablePosition($issuer, $r); }
                    if ($expectedKind === 'event') { $this->payablePosition($issuer,$this->object($issuer,'payable',(int)$r['payable_id'])); }
                    if ($expectedKind === 'receipt') { $this->received($issuer,$this->object($issuer,'order',(int)$r['order_id'])); }
                    if ($expectedKind === 'order') { $this->received($issuer,$r); }
                    if ($expectedKind === 'file') { $this->fileIntegrity($r); }
                    if ($this->portal!==null&&isset($r['request_id'])&&((int)$r['request_id']!==$this->portal['request_id']||($r['supplier_id']!==null&&(int)$r['supplier_id']!==$this->portal['supplier_id']))) { throw new DomainException('company_purchase_portal_denied'); }
                }
                $this->remember('company_purchase_commands', $old);
                return $result;
            }
            [$kind,$id,$result] = $work($issuer, $actor);
            $meta = ['kind'=>$kind,'id'=>$id,'origin'=>$this->portal === null ? ($this->api === null ? 'staff' : 'api') : 'supplier_portal'];
            if ($this->portal !== null) { $meta['invite_id'] = $this->portal['id']; $meta['supplier_id'] = $this->portal['supplier_id']; }
            $this->insert('issuers_audit', ['issuer_id'=>$issuer,'actor_id'=>$this->portal === null ? $actor : 0,'event'=>'company_purchase_'.$op,'subject_id'=>$id,'metadata_json'=>CompanyPurchasesValues::json($meta),'created_at'=>self::now()]);
            $resultJson=json_encode($result,JSON_THROW_ON_ERROR); $resultHash=hash('sha256',CompanyPurchasesValues::json([$issuer,$actor,$op,$hash,$kind,$id,$resultJson]));
            $this->insert('company_purchase_commands', ['issuer_id'=>$issuer,'actor_id'=>$actor,'request_key'=>$key,'operation'=>$op,'input_hash'=>$hash,'object_kind'=>$kind,'object_id'=>$id,'result_json'=>$resultJson,'result_hash'=>$resultHash,'created_at'=>self::now()]);
            return $result;
        });
    }

    private function settingsRow(int $issuer): array
    {
        $r = $this->rows('SELECT * FROM '.$this->p.'company_purchase_settings WHERE issuer_id=? FOR UPDATE', [$issuer])[0] ?? null;
        if (!$r) { throw new DomainException('company_purchase_settings_required'); }
        $this->remember('company_purchase_settings', $r, 'issuer_id'); return $r;
    }

    private function number(int $issuer, string $kind): array
    {
        $settings = $this->settingsRow($issuer); $table = self::table($kind);
        $number = (int)$this->rows('SELECT COALESCE(MAX(number),0) n FROM '.$this->p.$table.' WHERE issuer_id=?', [$issuer])[0]['n'] + 1;
        if ($number > 2147483647) { throw new DomainException('company_purchase_number_limit'); }
        $prefix = match ($kind) { 'request'=>'REQ','quote'=>'COT','order'=>'PED','receipt'=>'REC','document'=>'DOC',default=>throw new DomainException('company_purchase_invalid') };
        return [$number,$settings['prefix'].'-'.$prefix.'-'.$number];
    }

    private function immutable(array $data): array { return $data + ['record_hash'=>CompanyPurchasesValues::hash($data)]; }

    private function watchSet(string $table, array $where): void
    {
        $parts=[]; foreach ($where as $f=>$v) { $parts[]='`'.$f.'`=?'; }
        $sql='SELECT * FROM '.$this->p.$table.' WHERE '.implode(' AND ',$parts).' ORDER BY id';
        $args=array_values($where); $want=$this->rows($sql,$args);
        $this->validators[]=function () use ($sql,$args,$want): void { if ($this->rows($sql,$args)!==$want) { throw new RuntimeException('issuer_purchase_write_unconfirmed'); } };
    }
}
